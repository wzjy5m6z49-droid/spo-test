Option Explicit

'追加行対応
Private Const BEFORE_START_ROW As Long = 4
Private Const BEFORE_END_ROW As Long = 24
Private Const BEFORE_DONE_DATE_ROW As Long = 28
Private Const BEFORE_DONE_USER_ROW As Long = 29

Private Const AFTER_START_ROW As Long = 33
Private Const AFTER_END_ROW As Long = 54
Private Const AFTER_DONE_DATE_ROW As Long = 56
Private Const AFTER_DONE_USER_ROW As Long = 57

'***************************************
'* 20230405
'*
Sub valueCheck()
    On Error GoTo ErrHandler
    
    Dim ret As Boolean
    Dim i As Long
    Dim item As Long
    Dim ans As Long
    Dim userName As String

    userName = Split(Environ("USERNAME"), "(")(0)
    buttonName = Application.Caller
    
    If buttonName = "BeforeButton" Then
        
        If Cells(BEFORE_DONE_DATE_ROW, col.確認結果等) <> "" Then
            MsgBox "異動前チェックは完了しています。確認結果等の訂正はできません。" & vbCrLf & "※管理者による訂正は可能です。", vbCritical
            Exit Sub
        End If
        
        i = BEFORE_START_ROW
        ret = CheckFields(i)
        
        If ret = False Then Exit Sub
        
        ans = MsgBox("異動前チェックを行います。よろしいですか？" & vbCrLf & vbCrLf & _
            "【注意！】異動前チェック後の確認結果等の訂正はできません。" & vbCrLf & _
            "※管理者による訂正は可能です。", vbYesNo)
        
        If ans = vbNo Then
            MsgBox "確認処理を中止します。"
            Exit Sub
        End If
        
        Application.EnableEvents = False
        
        ActiveSheet.Unprotect Password:="0000"
        Cells(BEFORE_DONE_DATE_ROW, col.確認結果等).Value = Date
        Cells(BEFORE_DONE_USER_ROW, col.確認結果等).Value = userName
        Call LockFieldsBefore
        
        Application.EnableEvents = True
        
        DoEvents
        Application.Calculate
        
        MsgBox "異動前チェックが終了しました。確認結果等をロックします。", vbInformation
        
    Else
        
        If Cells(BEFORE_DONE_DATE_ROW, col.確認結果等) = "" Then
            MsgBox "異動前チェックが未了です。" & vbCrLf & "異動前チェック後に後任者確認をしてください。"
            Exit Sub
        End If
        
        If Cells(AFTER_DONE_DATE_ROW, col.確認結果等) <> "" Then
            MsgBox "後任者確認は完了しています。確認結果等の訂正はできません。" & vbCrLf & "※管理者による訂正は可能です。", vbCritical
            Exit Sub
        End If
        
        i = AFTER_START_ROW
        ret = CheckFields(i)
        
        If ret = False Then Exit Sub
        
        If CheckItem <> "" Then
            item = MsgBox(CheckItem() & _
                "の確認結果が異動前と異なります。" & vbCrLf & vbCrLf & _
                "処理を継続してもよろしいですか？", vbYesNo)
        End If
        
        If item = vbNo Then
            MsgBox "確認処理を中止します。"
            Exit Sub
        End If
        
        ans = MsgBox("後任者確認を行います。よろしいですか？" & vbCrLf & vbCrLf & _
            "【注意！】後任者確認後の確認結果等の訂正はできません。" & vbCrLf & _
            "※管理者による訂正は可能です。", vbYesNo)
        
        If ans = vbNo Then
            MsgBox "確認処理を中止します。"
            Exit Sub
        End If
        
        Application.EnableEvents = False
        
        ActiveSheet.Unprotect Password:="0000"
        Cells(AFTER_DONE_DATE_ROW, col.確認結果等).Value = Date
        Cells(AFTER_DONE_USER_ROW, col.確認結果等).Value = userName
        Call LockFieldsAfter
        
        Application.EnableEvents = True
        
        DoEvents
        Application.Calculate
        
        MsgBox "後任者確認が終了しました。確認結果等をロックします。", vbInformation
        
    End If
    
    Exit Sub

ErrHandler:
    Application.EnableEvents = True
    MsgBox "処理中にエラーが発生しました。" & vbCrLf & _
           "エラー番号：" & Err.Number & vbCrLf & _
           Err.Description, vbExclamation
End Sub

Public Function CheckFields(ByVal i As Long) As Boolean
    CheckFields = True
    
    Dim msg As String
    Dim cntRow As Long
    Dim endRow As Long
    
    If buttonName = "BeforeButton" Then
        endRow = BEFORE_END_ROW
    Else
        endRow = AFTER_END_ROW
    End If
    
    For cntRow = i To endRow
        
        'その他の機器入力チェック
        '異動前：15のみ
        If cntRow = 15 Then
            
            If Cells(cntRow, col.確認項目 + 1) <> "" Then
                If Cells(cntRow, col.確認結果等).Value = "" Then
                    msg = msg & "【項番" & Cells(cntRow, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
                    CheckFields = False
                End If
            Else
                If Cells(cntRow, col.確認結果等) = "有" Then
                    msg = msg & "【項番" & Cells(cntRow, col.項番) & "】のその他の機器を入力してください。" & vbCrLf
                    CheckFields = False
                End If
            End If
            
        '異動前チェック
        'G16もここに入れる
        ElseIf cntRow = 16 Or cntRow = 17 Or cntRow = 18 Or cntRow = 19 Or _
               cntRow = 20 Or cntRow = 21 Or cntRow = 22 Or _
               cntRow = 24 Then
            
            If Cells(cntRow, col.確認結果等).Value = "" Then
                msg = msg & "【項番" & Cells(cntRow, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
                CheckFields = False
            ElseIf Cells(cntRow, col.確認結果等) = "いいえ" Then
                msg = msg & "【項番" & Cells(cntRow, col.項番) & "】の処理に不備があります。確認項目のとおり処理を行い、確認結果等について再度選択してください。" & vbCrLf
                CheckFields = False
            End If
            
        '後任者確認：G45～G49
        ElseIf cntRow = 45 Or cntRow = 46 Or cntRow = 47 Or cntRow = 48 Or cntRow = 49 Then
            
            If Cells(cntRow, col.確認結果等).Value = "" Then
                msg = msg & "【項番" & Cells(cntRow, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
                CheckFields = False
            ElseIf Cells(cntRow, col.確認結果等) = "不可" Then
                msg = msg & "【項番" & Cells(cntRow, col.項番) & "】ログイン後、確認結果等を再度選択してください。" & vbCrLf
                CheckFields = False
            End If
            
        Else
            
            '25・26・51・52・53・54行目は通常の空欄チェックから除外
            If cntRow <> 25 And cntRow <> 26 And _
               cntRow <> 51 And cntRow <> 52 And _
               cntRow <> 53 And cntRow <> 54 Then
               
                If Cells(cntRow, col.確認結果等).Value = "" Then
                    msg = msg & "【項番" & Cells(cntRow, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
                    CheckFields = False
                End If
                
            End If
            
        End If
        
    Next
    
    '前任：G25に値がある場合、G26は空欄・いいえNG
    If buttonName = "BeforeButton" Then
        If Cells(25, col.確認結果等).Value <> "" Then
            
            If Cells(26, col.確認結果等).Value = "" Then
                msg = msg & "【項番" & Cells(26, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
                CheckFields = False
                
            ElseIf Cells(26, col.確認結果等).Value = "いいえ" Then
                msg = msg & "【項番" & Cells(26, col.項番) & "】の処理に不備があります。確認項目のとおり処理を行い、確認結果等について再度選択してください。" & vbCrLf
                CheckFields = False
            End If
            
        End If
    End If
    
    '後任：G51が空欄ならエラー
    If buttonName = "AfterButton" Then
    
        If Cells(51, col.確認結果等).Value = "" Then
            msg = msg & "【項番" & Cells(51, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
            CheckFields = False
        End If
        
        If Cells(52, col.確認結果等).Value = "" Then
            msg = msg & "【項番" & Cells(52, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
            CheckFields = False
            
        ElseIf Cells(52, col.確認結果等).Value = "いいえ" Then
            msg = msg & "【項番" & Cells(52, col.項番) & "】の処理に不備があります。確認項目のとおり処理を行い、確認結果等について再度選択してください。" & vbCrLf
            CheckFields = False
        End If
        
        If Cells(53, col.確認結果等).Value <> "" Then
            
            If Cells(54, col.確認結果等).Value = "" Then
                msg = msg & "【項番" & Cells(54, col.項番) & "】の確認結果等を入力してください。" & vbCrLf
                CheckFields = False
                
            ElseIf Cells(54, col.確認結果等).Value = "いいえ" Then
                msg = msg & "【項番" & Cells(54, col.項番) & "】の処理に不備があります。確認項目のとおり処理を行い、確認結果等について再度選択してください。" & vbCrLf
                CheckFields = False
            End If
            
        End If
        
    End If
        
    If CheckFields = False Then
        MsgBox msg, vbInformation
    End If
    
End Function
    
End Function

Public Function CheckItem() As String

    CheckItem = ""
    
    If buttonName = "AfterButton" Then
    
        Dim cntRow As Long
        
        '後任者側：37～44 ⇔ 異動前：8～15
        For cntRow = 37 To 44
        
            If Cells(cntRow, col.確認結果等).Value <> _
               Cells(cntRow - 29, col.確認結果等).Value Then
                
                CheckItem = CheckItem & _
                    "【項番" & Cells(cntRow, col.項番) & "】" & vbCrLf
                    
            End If
            
        Next
        
    End If

End Function

        MsgBox "確認結果のロックはすでに解除されています。"
        Exit Sub
    End If
    
    pwCheck = False
    パスワード認証.Show
    
    If pwCheck = True Then
        Call LiftFieldsBefore

    Range("A1").Select
End Sub

'20240605  USBルーターと無線LANアダプターのチェック追加
Private Sub Worksheet_Change(ByVal Target As Range)

    Dim g11 As Range
    Dim g12 As Range
    Dim g40 As Range
    Dim g41 As Range
    
    Set g11 = Me.Range("G11")
    Set g12 = Me.Range("G12")
    
    '後任者確認：G40/G41
    Set g40 = Me.Range("G40")
    Set g41 = Me.Range("G41")

    '異動前
    If Not Intersect(Target, g11) Is Nothing Or _
       Not Intersect(Target, g12) Is Nothing Then
       
        If g11.Value = "有" And g12.Value = "有" Then
            MsgBox "USB接続通信ルーターと無線LANアダプターの両方に【有】を選択することはできません。", vbExclamation
            
            g11.Value = ""
            g12.Value = ""
            
            Range("G40:G41").Value = ""
        End If
        
        ActiveCell.Select
    End If
    
    '後任者確認
    If Not Intersect(Target, g40) Is Nothing Or _
       Not Intersect(Target, g41) Is Nothing Then
       
        If g40.Value = "有" And g41.Value = "有" Then
            MsgBox "USB接続通信ルーターと無線LANアダプターの両方に【有】を選択することはできません。", vbExclamation
            
            Range("G40:G41").Value = ""
        End If
        
        ActiveCell.Select
    End If

End Sub
